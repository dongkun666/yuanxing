### 9.2 脚本内容 (概要)

```bash
#!/bin/bash
# scripts/trigger-paid-email-723.sh
# W14 first-paid-triggers-729 cron trigger
# 执行时间: 2026-07-23 09:00 CST
# 触发邮件: 5 律师 (L1-L5) 提前 6 天转化邮件

set -e

# 1. 读取律师邮箱 (placeholder, 公测前填实)
LAWYER_EMAILS=(
  "l1.founding@lexprime.cn"
  "l2.founding@lexprime.cn"
  "l3.founding@lexprime.cn"
  "l4.founding@lexprime.cn"
  "l5.founding@lexprime.cn"
)

# 2. 读取邮件模板 (first-paid-triggers-runbook.md § 3.1)
EMAIL_TEMPLATE="docs/marketing/first-paid-triggers-runbook.md"
EMAIL_SUBJECT="您的 LexPrime 试用即将到期 - 创史体验官特别优惠"

# 3. 触发邮件 (Mavis SMTP 本地)
for email in "${LAWYER_EMAILS[@]}"; do
  echo "[$(date +%H:%M:%S)] 触发邮件至 $email"
  python scripts/send_email.py \
    --to "$email" \
    --subject "$EMAIL_SUBJECT" \
    --body "$(cat $EMAIL_TEMPLATE | sed -n '/### 3.1/,/### 3.2/p')"
done

# 4. 写入 dashboard 触达日志
python scripts/log_trigger.py \
  --trigger-id "first-paid-triggers-729" \
  --trigger-time "2026-07-23T09:00:00+08:00" \
  --trigger-channel "email" \
  --trigger-target "5-lawyers" \
  --trigger-status "fire-success" \
  --log-file "docs/marketing/dashboard-paid-triggers.md"

echo "[$(date +%H:%M:%S)] 7/23 提前 6 天转化邮件触发完成 (5 律师)"
```
