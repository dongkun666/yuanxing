### 6.2 SQL 验证脚本 (每天 23:00 跑)

```sql
-- 触达率 (5 律师)
SELECT
  COUNT(DISTINCT user_id) AS reached_lawyers,
  5 AS total_lawyers,
  COUNT(DISTINCT user_id) / 5.0 * 100 AS reach_rate
FROM events
WHERE event IN ('paid_email_sent', 'wechat_1v1_sent', 'wechat_moments_sent')
  AND event_date BETWEEN '2026-07-23' AND '2026-07-30';

-- 转化率 (5 律师)
SELECT
  COUNT(DISTINCT user_id) AS converted_lawyers,
  5 AS total_lawyers,
  COUNT(DISTINCT user_id) / 5.0 * 100 AS conversion_rate
FROM events
WHERE event = 'paid_converted'
  AND invite_code IN ('BETA2025-0001', 'BETA2025-0002', 'BETA2025-0003', 'BETA2025-0004', 'BETA2025-0005')
  AND event_date BETWEEN '2026-07-29' AND '2026-07-30';

-- 营收 (5 律师)
SELECT
  SUM(amount) AS total_revenue,
```
